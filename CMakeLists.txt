cmake_minimum_required(VERSION 3.24)
project(Forge VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Keep the existing GPU build as the default; Mac development opts out explicitly.
option(FORGE_ENABLE_CUDA "Build NVIDIA CUDA operators, tests, and benchmarks" ON)
include(CTest)

add_library(forge_core src/tensor.cpp src/memory.cpp src/stream.cpp src/graph.cpp)
target_include_directories(forge_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

add_library(forge_ops src/ops/matmul.cpp src/ops/activations.cpp)
target_link_libraries(forge_ops PUBLIC forge_core)

add_library(forge_runtime src/runtime.cpp)
target_link_libraries(forge_runtime PUBLIC forge_ops)

if(BUILD_TESTING)
    add_executable(forge_tensor_test tests/tensor_test.cpp)
    target_link_libraries(forge_tensor_test PRIVATE forge_core)
    add_test(NAME tensor_metadata COMMAND forge_tensor_test)
    add_executable(forge_memory_test tests/memory_test.cpp)
    target_link_libraries(forge_memory_test PRIVATE forge_core)
    target_compile_definitions(forge_memory_test PRIVATE FORGE_TEST_CUDA=$<BOOL:${FORGE_ENABLE_CUDA}>)
    add_test(NAME memory_cpu COMMAND forge_memory_test)
    add_executable(forge_operator_validation_test tests/operator_validation_test.cpp)
    target_link_libraries(forge_operator_validation_test PRIVATE forge_ops)
    target_compile_definitions(forge_operator_validation_test PRIVATE FORGE_TEST_CUDA=$<BOOL:${FORGE_ENABLE_CUDA}>)
    add_test(NAME operator_validation COMMAND forge_operator_validation_test)
    add_executable(forge_activation_validation_test tests/activation_validation_test.cpp)
    target_link_libraries(forge_activation_validation_test PRIVATE forge_ops)
    target_compile_definitions(forge_activation_validation_test PRIVATE FORGE_TEST_CUDA=$<BOOL:${FORGE_ENABLE_CUDA}>)
    add_test(NAME activation_validation COMMAND forge_activation_validation_test)
    add_executable(forge_graph_test tests/graph_test.cpp)
    target_link_libraries(forge_graph_test PRIVATE forge_runtime)
    target_compile_definitions(forge_graph_test PRIVATE FORGE_TEST_CUDA=$<BOOL:${FORGE_ENABLE_CUDA}>)
    add_test(NAME graph_validation COMMAND forge_graph_test)
endif()

if(FORGE_ENABLE_CUDA)
    # Set this before enabling CUDA so compiler detection uses the intended target.
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES 75)
    endif()
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 17)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
    find_package(CUDAToolkit REQUIRED)
    target_compile_definitions(forge_core PRIVATE FORGE_HAS_CUDA=1)
    target_link_libraries(forge_core PRIVATE CUDA::cudart)

    add_library(forge_cuda cuda/vector_add.cu cuda/matmul.cu cuda/relu.cu cuda/softmax.cu)
    target_include_directories(forge_cuda PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cuda)
    target_link_libraries(forge_cuda PUBLIC CUDA::cudart)

    target_compile_definitions(forge_ops PRIVATE FORGE_HAS_CUDA=1)
    target_include_directories(forge_ops PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/cuda)
    target_link_libraries(forge_ops PRIVATE forge_cuda)

    add_executable(forge_vector_add benchmarks/vector_add_benchmark.cpp)
    target_link_libraries(forge_vector_add PRIVATE forge_cuda)
    add_executable(forge_matmul_benchmark benchmarks/matmul_benchmark.cpp)
    target_link_libraries(forge_matmul_benchmark PRIVATE forge_ops forge_cuda)

    add_executable(forge_inference examples/inference.cpp)
    target_link_libraries(forge_inference PRIVATE forge_runtime)

    if(BUILD_TESTING)
        add_executable(forge_matmul_test tests/matmul_test.cpp)
        target_link_libraries(forge_matmul_test PRIVATE forge_ops forge_cuda)
        add_test(NAME matmul_cuda COMMAND forge_matmul_test)
        add_executable(forge_runtime_test tests/runtime_test.cpp)
        target_link_libraries(forge_runtime_test PRIVATE forge_runtime)
        add_test(NAME runtime_cuda COMMAND forge_runtime_test)
        set_tests_properties(runtime_cuda PROPERTIES LABELS "gpu")
        add_executable(forge_activation_test tests/activation_test.cpp)
        target_link_libraries(forge_activation_test PRIVATE forge_ops forge_cuda)
        add_test(NAME activations_cuda COMMAND forge_activation_test)
        set_tests_properties(activations_cuda PROPERTIES LABELS "gpu")
        add_executable(forge_memory_cuda_test tests/memory_cuda_test.cpp)
        target_link_libraries(forge_memory_cuda_test PRIVATE forge_core CUDA::cudart)
        add_test(NAME memory_cuda COMMAND forge_memory_cuda_test)
        set_tests_properties(memory_cuda PROPERTIES LABELS "gpu")
        set_tests_properties(matmul_cuda PROPERTIES LABELS "gpu")
    endif()
endif()
